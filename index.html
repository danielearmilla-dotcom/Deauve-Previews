document.addEventListener('DOMContentLoaded', () => { // --- 1. MODO AMBIENTE REACTIVO (Canvas de orbes flotantes) --- const canvas = document.createElement('canvas'); canvas.id = 'ambient-canvas'; document.body.prepend(canvas); const ctx = canvas.getContext('2d'); let width = canvas.width = window.innerWidth; let height = canvas.height = window.innerHeight; window.addEventListener('resize', () => { width = canvas.width = window.innerWidth; height = canvas.height = window.innerHeight; }); const orbs = [ { x: width * 0.2, y: height * 0.2, radius: 180, color: 'rgba(255, 42, 133, 0.15)', vx: 0.4, vy: 0.2 }, { x: width * 0.8, y: height * 0.7, radius: 220, color: 'rgba(29, 185, 84, 0.08)', vx: -0.3, vy: -0.3 } ]; function animateAmbient() { ctx.clearRect(0, 0, width, height); orbs.forEach(orb => { orb.x += orb.vx; orb.y += orb.vy; if (orb.x < 0 || orb.x > width) orb.vx *= -1; if (orb.y < 0 || orb.y > height) orb.vy *= -1; const gradient = ctx.createRadialGradient(orb.x, orb.y, 0, orb.x, orb.y, orb.radius); gradient.addColorStop(0, orb.color); gradient.addColorStop(1, 'transparent'); ctx.fillStyle = gradient; ctx.beginPath(); ctx.arc(orb.x, orb.y, orb.radius, 0, Math.PI * 2); ctx.fill(); }); requestAnimationFrame(animateAmbient); } animateAmbient(); // --- 2. SELECTOR DE TEMA / PALETA DE COLORES POR CANCIÓN --- // Puedes asignar un color personalizado opcional por tarjeta con data-accent, o usar el rosa por defecto. const root = document.documentElement; function applySongTheme(card) { const accent = card.getAttribute('data-accent') || '#ff2a85'; root.style.setProperty('--pink', accent); root.style.setProperty('--pink-glow', `${accent}66`); // Actualizamos dinámicamente el color del orbe ambiental principal orbs[0].color = `${accent}26`; // ~15% opacidad en HEX } // --- 3. REPRODUCTOR Y VISUALIZADOR DE AUDIO REAL (Web Audio API) --- const songCards = document.querySelectorAll('.song-card'); let audioCtx = null; let analyser = null; let audioSourceNode = null; let dataArray = null; let currentCardPlaying = null; let animationFrameId = null; songCards.forEach(card => { const audio = card.querySelector('audio'); const playBtn = card.querySelector('.play-btn'); const waveform = card.querySelector('.waveform'); const timeText = card.querySelector('.time-text'); // Crear barra de progreso interna del waveform let progressFill = card.querySelector('.waveform-progress'); if (!progressFill) { progressFill = document.createElement('div'); progressFill.className = 'waveform-progress'; waveform.appendChild(progressFill); } // Formatear tiempo en MM:SS function formatTime(seconds) { if (isNaN(seconds)) return "0:00"; const mins = Math.floor(seconds / 60); const secs = Math.floor(seconds % 60); return `${mins}:${secs < 10 ? '0' : ''}${secs}`; } audio.addEventListener('loadedmetadata', () => { if(timeText && audio.duration) { timeText.textContent = formatTime(audio.duration); } }); // Inicializar Web Audio API al primer gesto del usuario function initWebAudioAPI(audioElement) { if (!audioCtx) { const AudioContext = window.AudioContext || window.webkitAudioContext; audioCtx = new AudioContext(); analyser = audioCtx.createAnalyser(); audioSourceNode = audioCtx.createMediaElementSource(audioElement); audioSourceNode.connect(analyser); analyser.connect(audioCtx.destination); analyser.fftSize = 64; dataArray = new Uint8Array(analyser.frequencyBinCount); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } } // Bucle del visualizador de frecuencia real function renderRealVisualizer() { if (!analyser || !currentCardPlaying) return; analyser.getByteFrequencyData(dataArray); const bars = currentCardPlaying.querySelectorAll('.playing-bars span'); bars.forEach((bar, index) => { const val = dataArray[index * 3] || 0; const heightPct = Math.max(15, (val / 255) * 100); bar.style.height = `${heightPct}%`; }); animationFrameId = requestAnimationFrame(renderRealVisualizer); } // Control del botón Play / Pause playBtn.addEventListener('click', () => { initWebAudioAPI(audio); // Si otra tarjeta está sonando, pausarla primero songCards.forEach(otherCard => { if (otherCard !== card) { const otherAudio = otherCard.querySelector('audio'); otherAudio.pause(); otherAudio.currentTime = 0; otherCard.classList.remove('playing', 'is-playing'); } }); if (audio.paused) { audio.play(); card.classList.add('playing', 'is-playing'); currentCardPlaying = card; applySongTheme(card); cancelAnimationFrame(animationFrameId); renderRealVisualizer(); } else { audio.pause(); card.classList.remove('playing', 'is-playing'); currentCardPlaying = null; cancelAnimationFrame(animationFrameId); } }); // Actualizar barra de progreso y tiempo actual audio.addEventListener('timeupdate', () => { if (audio.duration) { const percent = (audio.currentTime / audio.duration) * 100; progressFill.style.width = `${percent}%`; timeText.textContent = formatTime(audio.currentTime); } }); // Al terminar la pista audio.addEventListener('ended', () => { card.classList.remove('playing', 'is-playing'); progressFill.style.width = '0%'; timeText.textContent = formatTime(audio.duration); currentCardPlaying = null; cancelAnimationFrame(animationFrameId); }); // Scrubbing interactivo en la barra de onda (Waveform click) waveform.addEventListener('click', (e) => { const rect = waveform.getBoundingClientRect(); const clickX = e.clientX - rect.left; const widthCard = rect.width; const clickPercent = clickX / widthCard; if (audio.duration) { audio.currentTime = clickPercent * audio.duration; } }); }); });
